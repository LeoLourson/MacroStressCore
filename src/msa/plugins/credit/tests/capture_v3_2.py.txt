"""Capture numerical references of Track A v3.2 before the platform migration.

Runs every asset of DEMO-1 and DEMO-2 through the current ``run_asset`` on a scenario that
moves five drivers (revenue, cogs, interest, working capital, capex) through a graph with
lags and a triangular elasticity, and aggregates each portfolio. Writes the inputs and the
results to tests/golden/credit_v3_2.json.
"""

import hashlib
import json
from datetime import date
from pathlib import Path

import numpy as np

from msa.agents.aggregate import AggregateAgent, AggregateInput
from msa.agents.draw_macro import ScenarioDraws
from msa.agents.quant import QuantInput, review_result
from msa.contracts import AssetProfile, TransmissionGraph
from msa.engine.pipeline import AssetRiskInput, run_asset, scenario_set_from_parameters
from msa.engine.registry_store import DEMO_REGISTRY, load

ROOT = Path(__file__).resolve().parents[0]
REPO = Path(r"c:/Users/LeoLeon/Documents/МТС-Хакатон/macrostress-core")
AS_OF = date(2026, 9, 11)
SEED = 20260911
POINTS = 256
REPLICATES = 8
HORIZON = 12

EVIDENCE = [
    {
        "kind": "document",
        "source_id": "golden-event",
        "locator": "p1",
        "fragment_sha256": hashlib.sha256(b"golden").hexdigest(),
    }
]

CHANNELS = {
    "ch_demand": ("revenue", 0),
    "ch_costs": ("cogs", 1),
    "ch_rate": ("interest", 2),
    "ch_payments": ("working_capital", 0),
    "ch_capex": ("capex", 1),
}

SCENARIO = {
    "demand": {
        "kind": "triangular", "low": -0.18, "mode": -0.1, "high": -0.03,
        "unit": "fraction", "channel": "ch_demand", "driver": "revenue",
    },
    "costs": {
        "kind": "uniform", "low": 0.02, "high": 0.12,
        "unit": "fraction", "channel": "ch_costs", "driver": "cogs",
    },
    "rate": {
        "kind": "truncated_normal", "mean": 0.03, "sigma": 0.01, "low": 0.0, "high": 0.06,
        "unit": "fraction", "channel": "ch_rate", "driver": "interest",
    },
    "payments": {
        "kind": "triangular", "low": 5.0, "mode": 15.0, "high": 30.0,
        "unit": "days", "channel": "ch_payments", "driver": "working_capital",
    },
    "capex": {
        "kind": "point", "value": 0.1,
        "unit": "fraction", "channel": "ch_capex", "driver": "capex",
    },
}


def edge(src, dst, *, lag=0, elasticity=None):
    return {
        "src": src,
        "dst": dst,
        "sign": 1,
        "elasticity": elasticity or {"family": "point", "value": 1.0},
        "lag_months": lag,
        "evidence": EVIDENCE,
    }


def graph_for(okveds):
    nodes = [{"id": "event", "kind": "event", "label": "событие"}]
    edges = []
    for channel, (driver, lag) in CHANNELS.items():
        nodes.append({"id": channel, "kind": "channel", "label": channel})
        edges.append(edge("event", channel))
    for okved in sorted(okveds):
        sector = "s_" + okved.replace(".", "_")
        nodes.append({"id": sector, "kind": "sector", "label": okved, "okved": okved})
        for channel, (driver, lag) in CHANNELS.items():
            elasticity = (
                {"family": "triangular", "low": 0.5, "mode": 0.8, "high": 1.2}
                if driver == "revenue"
                else None
            )
            edges.append(edge(channel, sector, lag=lag, elasticity=elasticity))
            target = f"d_{okved.replace('.', '_')}_{driver}"
            if target not in {n["id"] for n in nodes}:
                nodes.append({"id": target, "kind": "driver", "label": driver, "driver": driver})
            if not any(e["src"] == sector and e["dst"] == target for e in edges):
                edges.append(edge(sector, target))
    return {"event_id": "event", "nodes": nodes, "edges": edges, "driver_elasticity": {}}


def summary(draws):
    values = np.asarray(draws, dtype=float)
    return {
        "count": int(values.size),
        "sum": float(values.sum()),
        "sum_sq": float((values**2).sum()),
        "head": [float(v) for v in values[:16]],
        "p05": float(np.percentile(values, 5)),
        "p50": float(np.percentile(values, 50)),
        "p95": float(np.percentile(values, 95)),
    }


def main():
    registry = load(DEMO_REGISTRY)
    out = {
        "note": "Track A v3.2 results captured from macrostress-core before the platform "
        "migration (2026-09-25). Inputs are stored here; the reference test rebuilds them.",
        "as_of": AS_OF.isoformat(),
        "seed": SEED,
        "points": POINTS,
        "replicates": REPLICATES,
        "horizon": HORIZON,
        "registry_id": DEMO_REGISTRY,
        "scenario": SCENARIO,
        "portfolios": {},
    }
    for portfolio_id in ("DEMO-1", "DEMO-2"):
        catalog = json.loads(
            (REPO / "demo" / "portfolios" / f"{portfolio_id}.json").read_text("utf-8")
        )
        profiles = [AssetProfile.model_validate(a) for a in catalog["assets"]]
        graph = graph_for({p.okved for p in profiles})
        scenario = scenario_set_from_parameters(SCENARIO, points=POINTS, replicates=REPLICATES)
        results, draws = [], {}
        sampled = scenario.sample(SEED)
        for profile in profiles:
            if profile.status == "needs_review":
                # As QuantAgent: an incomplete profile stays in the run without numbers.
                result = review_result(
                    QuantInput.model_construct(
                        profile=profile,
                        draws=ScenarioDraws(
                            seed=SEED,
                            replicates=REPLICATES,
                            points=POINTS,
                            names=list(sampled.names),
                            sha256=sampled.sha256,
                        ),
                        as_of=AS_OF,
                        horizon=HORIZON,
                    ),
                    "profile_needs_review:" + ",".join(profile.flags),
                ).result
                results.append(result)
                continue
            result = run_asset(
                AssetRiskInput(profile, TransmissionGraph.model_validate(graph), AS_OF, HORIZON),
                scenario,
                registry.model,
                SEED,
            )
            results.append(result)
            if result.aggregation_draws is not None:
                draws[result.asset_id] = list(result.aggregation_draws)
        portfolio = AggregateAgent(registry_id=DEMO_REGISTRY).run(
            AggregateInput(
                run_id=f"golden-{portfolio_id}",
                results=results,
                loss_draws=draws,
                expected_ead={p.asset_id: float(p.ead) for p in profiles},
                macro_draws_sha256=results[0].macro_draws_sha256,
            )
        )
        out["portfolios"][portfolio_id] = {
            "graph": graph,
            "assets": {
                r.asset_id: {
                    "dq": next(float(p.dq) for p in profiles if p.asset_id == r.asset_id),
                    "result": r.model_dump(mode="json"),
                    "loss_draws": summary(draws[r.asset_id]) if r.asset_id in draws else None,
                }
                for r in results
            },
            "portfolio": portfolio.model_dump(mode="json"),
        }
        print(portfolio_id, [(r.asset_id, r.status) for r in results])
        print("  portfolio", portfolio.status, portfolio.loss_quantiles)
    target = REPO / "tests" / "golden" / "credit_v3_2.json"
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(json.dumps(out, ensure_ascii=False, indent=1, sort_keys=True), "utf-8")
    print("written", target, target.stat().st_size)


if __name__ == "__main__":
    main()
